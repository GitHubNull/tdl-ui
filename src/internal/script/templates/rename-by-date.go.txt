// 【功能】按消息日期归档并重命名下载文件，输出「年/年-月/对话ID_消息ID_原名」形式的相对路径。
// 【参数】subDirLayout：目录层级模板，支持 {{year}} {{month}} {{day}} 占位符；keepOriginalName：是否保留原文件名（false 时仅用 对话ID_消息ID + 扩展名）。
// 【注意】Rename 返回相对下载目录的路径，用 / 分隔；返回空串则回退设置页的全局命名模板。Yaegi 沙箱内仅可用 tdlui/api 与部分标准库，单次调用有超时保护，勿做阻塞操作。
package main

import (
	"fmt"
	"path"
	"strings"
	"time"

	"tdlui/api"
)

// ---- 可调参数 ----
var (
	// subDirLayout 子目录层级模板
	subDirLayout = "{{year}}/{{year}}-{{month}}"
	// keepOriginalName 是否在新名字里保留原始文件名
	keepOriginalName = true
)

// Rename 自定义下载文件名；返回空串表示回退默认模板。
func Rename(f api.FileInfo) string {
	t := time.Unix(f.MessageDate, 0)
	dir := subDirLayout
	dir = strings.ReplaceAll(dir, "{{year}}", t.Format("2006"))
	dir = strings.ReplaceAll(dir, "{{month}}", t.Format("01"))
	dir = strings.ReplaceAll(dir, "{{day}}", t.Format("02"))

	name := fmt.Sprintf("%d_%d", f.DialogID, f.MessageID)
	if keepOriginalName {
		name = name + "_" + sanitize(f.FileName)
	} else {
		name = name + path.Ext(f.FileName)
	}

	out := path.Join(dir, name)
	api.Logf("[归档重命名] %s -> %s", f.FileName, out)
	return out
}

// sanitize 替换文件名中不适合作为路径的字符。
func sanitize(name string) string {
	for _, bad := range []string{"\\", "/", ":", "*", "?", "\"", "<", ">", "|"} {
		name = strings.ReplaceAll(name, bad, "_")
	}
	return strings.TrimSpace(name)
}
